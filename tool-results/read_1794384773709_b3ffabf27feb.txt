    53→  // ---------- 2. Animated number counters ----------
    54→  function animateCounter(el) {
    55→    var target = parseFloat(el.dataset.count || "0");
    56→    var suffix = el.dataset.suffix || "";
    57→    var prefix = el.dataset.prefix || "";
    58→    var decimals = parseInt(el.dataset.decimals || "0", 10);
    59→    var duration = fxNone ? 0 : 1500;
    60→    var start = performance.now();
    61→    function fmt(v) {
    62→      return prefix + v.toFixed(decimals).replace(/\B(?=(\d{3})+(?!\d))/g, ",") + suffix;
    63→    }
    64→    function step(now) {
    65→      var t = Math.min(1, (now - start) / duration);
    66→      // easeOutCubic
    67→      var eased = 1 - Math.pow(1 - t, 3);
    68→      el.textContent = fmt(target * eased);
    69→      if (t < 1) requestAnimationFrame(step);
    70→      else el.textContent = fmt(target);
    71→    }
    72→    requestAnimationFrame(step);
    73→  }

   349→  // ---------- H2.b: split-text mask headline ----------
   350→  // Word-splits the hero headline in place (DOM APIs only — no innerHTML,
   351→  // same rule as the quote reveal) so each word can rise out of its own
   352→  // overflow-hidden mask. Ownership hand-over: once the words own the
   353→  // entrance, the h1's own hv2-rise fade is removed (a one-shot transition
   354→  // must never share the animated properties with the mask animation —
   355→  // same sole-owner rule as G3/D-LOOP-31a). fxNone skips the split entirely:
   356→  // the h1 stays plain and fully visible (D-MOTION-SAFE).
   357→  function setupHeroHeadline() {
   358→    var headline = document.querySelector("[data-hero-headline]");
   359→    if (!headline) return;
   360→    if (fxNone) return;
   361→    var index = 0;
   362→    // Split ONE text node inside its parent element: every run of
   363→    // non-whitespace becomes a mask span, every whitespace run is kept
   364→    // VERBATIM as plain text (leading/trailing/between — so the inter-part
   365→    // spaces of the template survive and innerText matches the plain h1).
   366→    function splitText(parent, textNode) {
   367→      var raw = textNode.textContent;
   368→      var frag = document.createDocumentFragment();
   369→      var re = /\S+/g;
   370→      var m;
   371→      var last = 0;
   372→      while ((m = re.exec(raw))) {
   373→        if (m.index > last) {
   374→          frag.appendChild(document.createTextNode(raw.slice(last, m.index)));
   375→        }
   376→        var mask = document.createElement("span");
   377→        mask.className = "hero__w";
   378→        var inner = document.createElement("span");
   379→        inner.className = "hero__w-in";
   380→        inner.style.setProperty("--d", index++);
   381→        inner.textContent = m[0];
   382→        mask.appendChild(inner);
   383→        frag.appendChild(mask);
   384→        last = m.index + m[0].length;
   385→      }
   386→      if (last < raw.length) {
   387→        frag.appendChild(document.createTextNode(raw.slice(last)));
   388→      }
   389→      parent.replaceChild(frag, textNode);
   390→    }
   391→    Array.prototype.slice.call(headline.childNodes).forEach(function (n) {
   392→      if (n.nodeType === 3 && n.textContent.trim()) {
   393→        splitText(headline, n);
   394→      } else if (n.nodeType === 1) {
   395→        Array.prototype.slice.call(n.childNodes).forEach(function (m) {
   396→          if (m.nodeType === 3 && m.textContent.trim()) splitText(n, m);
   397→        });
   398→      }
   399→    });
   400→    headline.classList.remove("hv2-rise");
   401→    headline.style.removeProperty("--d");
   402→  }

   477→  // ---------- r47: bubbly section-title reveal (data-bubble-title) ----------
   478→  // Every topic title pops in letter-by-letter (the bt-pop bubble in
   479→  // home.css: rise -> overshoot -> wobble -> settle) and REPLAYS when
   480→  // it re-enters view, scrolling down or back up (owner request).
   481→  // Hysteresis: pop at >=30% visible, re-arm only once the title has
   482→  // FULLY left (ratio 0) — a title drifting past the edge never
   483→  // flickers. Words wrap as units (.bt-w), letters animate (.bt-l);
   484→  // per-letter delay = word stagger (capped) + letter stagger, both
   485→  // set as CSS vars by the split.
   486→  // D-MOTION-SAFE: fxNone / no-JS / no-IO -> the title never splits
   487→  // and stays plain text. The .bt-armed class (the ONLY thing that
   488→  // hides letters) is added per element AFTER io.observe succeeds —
   489→  // a crash anywhere earlier leaves fully visible text behind.
   490→  // Split is DOM-APIs-only (S11): template text is never re-parsed.
   491→  function setupBubbleTitles() {
   492→    var els = document.querySelectorAll("[data-bubble-title]");
   493→    if (!els.length) return;
   494→    if (fxNone || !("IntersectionObserver" in window)) return;
   495→
   496→    function splitTitle(el) {
   497→      var wordIndex = 0;
   498→      Array.prototype.slice.call(el.childNodes).forEach(function (n) {
   499→        if (n.nodeType !== 3 || !n.textContent.trim()) return;
   500→        var raw = n.textContent;
   501→        var frag = document.createDocumentFragment();
   502→        var re = /\S+/g, m, last = 0;
   503→        while ((m = re.exec(raw))) {
   504→          if (m.index > last) {
   505→            frag.appendChild(document.createTextNode(raw.slice(last, m.index)));
   506→          }

   585→  // ---------- r69 (owner round thirteen): listings palm-field drift ----------
   586→  // Owner: "the palms are stagnant they need to move ...but not to slow
   587→  // they look stagnant and not to fast they move together with the
   588→  // listings." The r68 pass pinned the field to the viewport
   589→  // (background-attachment: fixed — speed 0, stagnant; iOS/Android
   590→  // ignore fixed attachment outright, so phones never saw even that).
   591→  // A plain scrolling background runs 1:1 with the rows — glued to
   592→  // the listings. HALF SPEED is the middle the owner described: the
   593→  // field visibly climbs as the page scrolls while the rows, chips
   594→  // and pagination slide past twice as fast.
   595→  //
   596→  // Mechanism: this writes --palms-y on .listings-band; search.css
   597→  // points the ::before's background-position at var(--palms-y, 0px).
   598→  // The tile repeats, so the drift can run forever without a seam,
   599→  // and the box-relative 84% mask stays anchored to the band — the
   600→  // dissolve still lands exactly at the footer line wherever the
   601→  // scroll is. The chase is LERPED, not snapped: during a fast fling
   602→  // the field hangs back a touch and eases in once the scroll
   603→  // settles — weight, not lag.
   604→  //
   605→  // D-MOTION-SAFE / D-FX-GATE: fxNone (fx=none OR the attribute
   606→  // missing — scripts blocked, fails closed) returns before anything
   607→  // moves; the gate owns the reduced-motion decision, no media query
   608→  // is re-checked here. With JS blocked the var keeps its 0px
   609→  // fallback: the exact static field. Decorative only — no element
   610→  // moves, nothing for assistive tech to notice.
   611→  function setupPalmParallax() {
   612→    if (fxNone) return;
   613→    var band = document.querySelector(".listings-band");
   614→    if (!band) return;
   615→    var DRIFT = 0.5;   // pattern px per scrolled px -> field speed 1 - 0.5 = HALF
   616→    var EASE = 0.14;   // lerp per frame — the weight in the chase
   617→    var current = 0;
   618→    var target = 0;
   619→    var raf = 0;
   620→
   621→    function measure() {
   622→      // An htmx swap (filters, pagination, sort) can replace the band
   623→      // wholesale — re-resolve it, and restart the chase from the
   624→      // static pose so a fresh field eases in instead of snapping.
   625→      if (!band.isConnected) {
   626→        band = document.querySelector(".listings-band");
   627→        current = 0;
   628→        if (!band) return;
   629→      }
   630→      var y = window.scrollY || window.pageYOffset || 0;
   631→      var rect = band.getBoundingClientRect();
   632→      var vh = window.innerHeight ||
   633→               document.documentElement.clientHeight;
   634→      // anchored at the scroll where the band's top enters the
   635→      // viewport: the field is exactly static on arrival, drifts
   636→      // while any part of the band is on screen, freezes after it
   637→      // has passed (no wasted work in the footer).
   638→      var entry = rect.top + y - vh;
   639→      var travel = Math.max(0, Math.min(y - entry, rect.height + 2 * vh));
   640→      target = travel * DRIFT;
   641→    }
   642→
   643→    function frame() {
   644→      raf = 0;
   645→      if (!band.isConnected) return;   // the next kick() re-resolves
   646→      current += (target - current) * EASE;
   647→      if (Math.abs(target - current) < 0.05) current = target;
   648→      band.style.setProperty("--palms-y", current.toFixed(2) + "px");
   649→      if (current !== target) raf = requestAnimationFrame(frame);
   650→    }
   651→
   652→    function kick() {
   653→      measure();
   654→      if (!band || current === target) return;
   655→      if (!raf) raf = requestAnimationFrame(frame);
   656→    }
   657→
   658→    window.addEventListener("scroll", kick, { passive: true });
   659→    window.addEventListener("resize", kick, { passive: true });
   660→    document.addEventListener("htmx:afterSwap", kick);
   661→    kick();
   662→  }
