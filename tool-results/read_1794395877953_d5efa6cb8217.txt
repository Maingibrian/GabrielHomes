     2→

    29→def _resolve_location(location_slug: str | None) -> Location | None:
    30→    """Resolve a location slug to a Location, matching at any level."""
    31→    if not location_slug:
    32→        return None
    33→    # Slug is unique per parent, so a bare slug may match multiple
    34→    # locations at different levels (e.g. "nyali" the area vs a
    35→    # neighbourhood that happens to share the slug). Prefer the
    36→    # shallowest match (county > area > neighbourhood).
    37→    qs = Location.objects.filter(slug=location_slug).order_by("level")
    38→    return qs.first()
    39→
    40→
    41→def _location_filter_q(location: Location) -> Q:
    42→    """Build a Q object matching the location AND all its descendants.
    43→
    44→    WORK.md §9: "searching 'Nairobi' includes Westlands, Kilimani, etc."
    45→    We walk 3 levels deep (county > area > neighbourhood) and OR the
    46→    pks together.
    47→    """
    48→    descendant_pks = set(location.get_descendants(include_self=True).values_list("pk", flat=True))
    49→    return Q(location__in=descendant_pks)

    52→def _build_h1(listing_type: str, ptype: PropertyType | None, location: Location | None) -> str:
    53→    """Generate the H1 from the URL context.
    54→
    55→    Examples:
    56→        sale, Flats & Apartments, None      -> "Flats & Apartments for Sale"
    57→        rent, Houses, Mombasa               -> "Houses for Rent in Mombasa"
    58→        sale, None, Mombasa                 -> "Property for Sale in Mombasa"
    59→    """
    60→    type_name = ptype.name if ptype else "Property"
    61→    if listing_type is None:
    62→        # Combined browse-all page (/properties/) — no type filter applied.
    63→        if not ptype:
    64→            type_name = "Properties"
    65→        return f"{type_name} for Sale and Rent" + (
    66→            f" in {location.name}" if location else ""
    67→        )
    68→    action = "for Sale" if listing_type == "sale" else "for Rent"
    69→    h1 = f"{type_name} {action}"
    70→    if location:
    71→        # Use the location's own name (not the full path) for the H1.
    72→        h1 += f" in {location.name}"
    73→    return h1

    76→def _build_breadcrumbs(listing_type: str, ptype: PropertyType | None, location: Location | None) -> list[dict]:
    77→    """Build breadcrumb trail: Home > [Type] for [Sale/Rent] > [Location]."""
    78→    crumbs = [{"name": "Home", "url": "/"}]
    79→    if listing_type is None:
    80→        # Combined page: Home > Properties > [Location]
    81→        crumbs.append({"name": "Properties", "url": ""})
    82→        if location:
    83→            crumbs.append({"name": location.name, "url": ""})
    84→        return crumbs
    85→    action = "for Sale" if listing_type == "sale" else "for Rent"
    86→    type_name = ptype.name if ptype else "Property"
    87→
    88→    if ptype:
    89→        type_url = f"/{ptype.slug_plural}-{'for-sale' if listing_type == 'sale' else 'for-rent'}/"
    90→    else:
    91→        type_url = f"/property-{'for-sale' if listing_type == 'sale' else 'for-rent'}/"
    92→
    93→    crumbs.append({"name": f"{type_name} {action}", "url": type_url})
    94→    if location:
    95→        crumbs.append({"name": location.name, "url": ""})  # current page
    96→    return crumbs

    99→def _build_canonical_url(listing_type: str, ptype: PropertyType | None, location: Location | None) -> str:
   100→    """Canonical URL = the clean path without query params (per §7)."""
   101→    if listing_type is None:
   102→        # Combined browse-all page.
   103→        return "/properties/" + (f"{location.slug}/" if location else "")
   104→    if ptype:
   105→        base = f"/{ptype.slug_plural}-{'for-sale' if listing_type == 'sale' else 'for-rent'}/"
   106→    else:
   107→        base = f"/property-{'for-sale' if listing_type == 'sale' else 'for-rent'}/"
   108→    if location:
   109→        base += f"{location.slug}/"
   110→    return base

   113→def _build_related_searches(listing_type: str, ptype: PropertyType | None, location: Location | None) -> list[dict]:
   114→    """Build a 'Related searches' link block for the bottom of the page."""
   115→    if listing_type is None:
   116→        # Combined page — point people at the two canonical SEO surfaces
   117→        # plus the top counties.
   118→        related = [
   119→            {"name": "Property for Sale", "url": "/property-for-sale/"},
   120→            {"name": "Property for Rent", "url": "/property-for-rent/"},
   121→        ]
   122→        top_counties = Location.objects.filter(
   123→            level=Location.Level.COUNTY
   124→        ).order_by("name")[:6]
   125→        for county in top_counties:
   126→            related.append({
   127→                "name": f"Property for Sale in {county.name}",
   128→                "url": f"/property-for-sale/{county.slug}/",
   129→            })
   130→        return related[:8]
   131→    related = []
   132→    action_suffix = "for-sale" if listing_type == "sale" else "for-rent"
   133→    action_label = "for Sale" if listing_type == "sale" else "for Rent"
   134→
   135→    if location:
   136→        # Other types in the same location
   137→        other_types = PropertyType.objects.exclude(pk=ptype.pk if ptype else None).order_by("position")[:5]
   138→        for ot in other_types:
   139→            related.append({
   140→                "name": f"{ot.name} {action_label} in {location.name}",
   141→                "url": f"/{ot.slug_plural}-{action_suffix}/{location.slug}/",
   142→            })
   143→        # Same type (or all-types) in sibling locations
   144→        if location.parent:
   145→            siblings = Location.objects.filter(parent=location.parent).exclude(pk=location.pk)[:4]
   146→            for sib in siblings:
   147→                if ptype:
   148→                    url = f"/{ptype.slug_plural}-{action_suffix}/{sib.slug}/"
   149→                else:
   150→                    url = f"/property-{action_suffix}/{sib.slug}/"
   151→                label_pt = ptype.name if ptype else "Property"
   152→                related.append({
   153→                    "name": f"{label_pt} {action_label} in {sib.name}",
   154→                    "url": url,
   155→                })
   156→    else:
   157→        # No location — show top locations for this type
   158→        top_counties = Location.objects.filter(level=Location.Level.COUNTY).order_by("name")[:5]
   159→        for county in top_counties:
   160→            if ptype:
   161→                url = f"/{ptype.slug_plural}-{action_suffix}/{county.slug}/"
   162→            else:
   163→                url = f"/property-{action_suffix}/{county.slug}/"
   164→            label_pt = ptype.name if ptype else "Property"
   165→            related.append({
   166→                "name": f"{label_pt} {action_label} in {county.name}",
   167→                "url": url,
   168→            })
   169→
   170→    return related[:8]

   173→def search(request, type_slug: str | None = None, location_slug: str | None = None, listing_type: str = "sale"):
   174→    """Search results page — the heart of the portal.
   175→
   176→    URL patterns (per §7):
   177→        /<type>-for-sale/                       -> type_slug set, no location
   178→        /<type>-for-sale/<location>/            -> both set
   179→        /<type>-for-rent/                       -> type_slug set, no location
   180→        /<type>-for-rent/<location>/            -> both set
   181→        /property-for-sale/<location>/          -> type_slug=None, location set
   182→        /property-for-rent/<location>/          -> type_slug=None, location set
   183→    """
   184→    # ------------------------------------------------------------------
   185→    # Resolve URL context
   186→    # ------------------------------------------------------------------
   187→    # No-JS hero-search fallback: when JavaScript is unavailable the hero
   188→    # form submits GET params (listing_type / type / location_slug) to the
   189→    # bare /property-for-<sale|rent>/ route. Honour those params only when
   190→    # the URL path didn't already provide them, so SEO URLs are unaffected.
   191→    if type_slug is None:
   192→        type_slug = request.GET.get("type") or None
   193→    if location_slug is None:
   194→        location_slug = request.GET.get("location_slug") or None
   195→
   196→    ptype = None
   197→    if type_slug:
   198→        ptype = get_object_or_404(PropertyType, slug_plural=type_slug)
   199→
   200→    location = _resolve_location(location_slug)
   201→    if location_slug and not location:
   202→        from django.http import Http404
   203→        raise Http404("Location not found.")
   204→
   205→    # ------------------------------------------------------------------
   206→    # Base queryset — only published listings, filtered by URL context
   207→    # ------------------------------------------------------------------
   208→    qs = Property.objects.published()
   209→    if listing_type:  # None on the combined /properties/ page — all types
   210→        qs = qs.filter(listing_type=listing_type)
   211→    else:
   212→        # Keep placeholder inventory Mombasa-only while never hiding genuine
   213→        # owner-entered listings should the business expand later.
   214→        mombasa = Location.objects.filter(level=Location.Level.COUNTY, name="Mombasa").first()
   215→        if mombasa:
   216→            qs = qs.filter(
   217→                Q(is_demo=False)
   218→                | Q(location__in=mombasa.get_descendants(include_self=True))
   219→            )
   220→    market = request.GET.get("market", "").lower()
   221→    if market == "sale":
   222→        qs = qs.filter(listing_type=Property.ListingType.SALE)
   223→    elif market == "rent":
   224→        qs = qs.filter(listing_type=Property.ListingType.RENT, is_short_stay=False)
   225→    elif market == "bnb":
   226→        qs = qs.filter(listing_type=Property.ListingType.RENT, is_short_stay=True)
   227→
   228→    if ptype:
   229→        qs = qs.filter(property_type=ptype)
   230→
   231→    if location:
   232→        qs = qs.filter(_location_filter_q(location))
   233→
   234→    # Eager-load everything the listing card needs (avoid N+1).
   235→    qs = qs.select_related(
   236→        "property_type", "location__parent__parent", "agent__user",
   237→    ).prefetch_related("images", "amenities")
   238→
   239→    # ------------------------------------------------------------------
   240→    # Apply query-param filters via the FilterSet
   241→    # ------------------------------------------------------------------
   242→    filter_set = PropertyFilter(request.GET, queryset=qs)
   243→    qs = filter_set.qs
   244→
   245→    # Keep the map readable: presentation/demo mode pins the curated homes,
   246→    # while every genuine owner-entered listing remains eligible.
   247→    map_source = qs.filter(Q(is_demo=False) | ~Q(address_text=""))
   248→    map_rows = map_source.exclude(lat__isnull=True).exclude(lng__isnull=True).values(
   249→        "lat", "lng", "title", "slug", "pk", "price", "listing_type", "is_short_stay", "location__name",
   250→        "location__parent__name", "location__parent__parent__name",
   251→    )[:18]
   252→    map_listings = []
   253→    for item in map_rows:
   254→        location_bits = [
   255→            item.get("location__parent__parent__name"),
   256→            item.get("location__parent__name"),
   257→            item.get("location__name"),
   258→        ]
   259→        map_listings.append({
   260→            "id": item["pk"],  # L1: pairs each pin with its result row
   261→            "lat": float(item["lat"]), "lng": float(item["lng"]),
   262→            "title": item["title"],
   263→            "url": f"/listing/{item['slug']}-{item['pk']}/",
   264→            "price": f"KES {item['price']:,.0f}",
   265→            "price_short": (
   266→                f"KSh {item['price'] / 1_000_000:.1f}M" if item["price"] >= 1_000_000
   267→                else f"KSh {item['price'] / 1_000:.0f}K"
   268→            ),
   269→            "market": "bnb" if item["is_short_stay"] else item["listing_type"],
   270→            "location": " > ".join(bit for bit in location_bits if bit),
   271→        })
   272→
   273→    # ------------------------------------------------------------------
   274→    # Paginate (24 per page)
   275→    # ------------------------------------------------------------------
   276→    paginator = Paginator(qs, PAGE_SIZE)
   277→    page_num = request.GET.get("page", "1")
   278→    try:
   279→        page = paginator.page(page_num)
   280→    except (PageNotAnInteger, EmptyPage):
   281→        page = paginator.page(1)
   282→
   283→    # ------------------------------------------------------------------
   284→    # Context
   285→    # ------------------------------------------------------------------
   286→    h1 = _build_h1(listing_type, ptype, location)
   287→    if listing_type is None and not location:
   288→        h1 = {
   289→            "sale": "Mombasa homes for sale",
   290→            "rent": "Mombasa homes for long-term rent",
   291→            "bnb": "Mombasa short stays and BnB",
   292→        }.get(market, "Homes across Mombasa")
   293→    breadcrumbs = _build_breadcrumbs(listing_type, ptype, location)
   294→    canonical_url = _build_canonical_url(listing_type, ptype, location)
   295→    related_searches = _build_related_searches(listing_type, ptype, location)
   296→
   297→    # Count of total results (before pagination)
   298→    total_count = paginator.count
   299→
   300→    # Whether we have any active filters (for the "Clear filters" link)
   301→    active_filters = bool(request.GET.dict())
   302→
   303→    context = {
   304→        "body_class": "page-search",
   305→        "page_title": h1,
   306→        "meta_description": f"{h1}. Browse checked listings on Gabriel Homes — real estate with a human point of view.",
   307→        "h1": h1,
   308→        "breadcrumbs": breadcrumbs,
   309→        "canonical_url": canonical_url,
   310→        "listing_type": listing_type,
   311→        "property_type": ptype,
   312→        "location": location,
   313→        "page_obj": page,
   314→        "total_count": total_count,
   315→        "page_size": PAGE_SIZE,
   316→        "filter_form": filter_set.form,
   317→        "filterset": filter_set,
   318→        "sort_options": SORT_OPTIONS,
   319→        "current_sort": request.GET.get("sort", "newest"),
   320→        "active_filters": active_filters,
   321→        "current_market": market,
   322→        "map_listings": map_listings,
   323→        "related_searches": related_searches,
   324→        "is_htmx": request.headers.get("HX-Request") == "true",
   325→        # For the filter sidebar — pass all property types and top locations
   326→        "all_property_types": PropertyType.objects.order_by("position", "name"),
   327→        "all_counties": Location.objects.filter(level=Location.Level.COUNTY).order_by("name"),
   328→    }
   329→
   330→    # If HTMX request, return just the results partial
   331→    if context["is_htmx"]:
   332→        return render(request, "listings/_results_grid.html", context)
   333→
   334→    return render(request, "listings/search.html", context)

   337→def location_autocomplete(request):
   338→    """JSON endpoint for the hero search bar's location input.
   339→
   340→    Returns up to 8 locations whose name starts with (or contains)
   341→    the `q` query param. Each result includes the slug, name, full
   342→    path, and level — enough for the JS to build a URL.
   343→
   344→    Usage: /search/locations/?q=mom
   345→    Returns: [{"slug": "mombasa", "name": "Mombasa", "full_path": "Mombasa", "level": "county"}, ...]
   346→    """
   347→    q = request.GET.get("q", "").strip()
   348→    if len(q) < 2:
   349→        return JsonResponse({"results": []})
   350→
   351→    # Match by name (case-insensitive icontains) — shallowest first.
   352→    qs = Location.objects.filter(name__icontains=q).order_by("level", "name")[:8]
   353→    results = [
   354→        {
   355→            "slug": loc.slug,
   356→            "name": loc.name,
   357→            "full_path": loc.full_path(),
   358→            "level": loc.level,
   359→        }
   360→        for loc in qs
   361→    ]
   362→    return JsonResponse({"results": results})

   424→def listing_detail(request, slug: str, pk: int):
   425→    """Property detail page — per WORK.md §8 (Listing detail).
   426→
   427→    URL: /listing/<slug>-<pk>/
   428→    - Only published listings are visible (404 otherwise).
   429→    - View counter is debounced per session (one count per session per
   430→      listing — a session flag prevents reloads from inflating the count).
   431→    - Sticky brand contact card (Call / WhatsApp / Email, brand row) and enquiry form.
   432→    - Save and Share buttons.
   433→    - Similar listings: same property_type + same county (via location
   434→      ancestors), ±30% price, exclude self.
   435→    - JSON-LD structured data for SEO.
   436→    - Leaflet map with approximate pin (respects address_text privacy).
   437→    """
   438→    from django.shortcuts import get_object_or_404, render as _render
   439→
   440→    property = get_object_or_404(
   441→        Property.objects.published().select_related(
   442→            "property_type",
   443→            "location__parent__parent",
   444→            "agent__user",
   445→        ).prefetch_related("images", "amenities"),
   446→        slug=slug, pk=pk,
   447→    )
   448→
   449→    # ------------------------------------------------------------------
   450→    # View counter — debounced per session (one count per session per listing)
   451→    # ------------------------------------------------------------------
   452→    session_key = f"viewed_property_{property.pk}"
   453→    if not request.session.get(session_key):
   454→        property.increment_views()
   455→        request.session[session_key] = True
   456→        # Mark session as modified so it's saved.
   457→        request.session.modified = True
   458→
   459→    # ------------------------------------------------------------------
   460→    # Similar listings — same type + same county, ±30% price, exclude self
   461→    # ------------------------------------------------------------------
   462→    from decimal import Decimal
   463→    price_min = property.price * Decimal("0.7")
   464→    price_max = property.price * Decimal("1.3")
   465→    # Get the county ancestor of this listing's location.
   466→    county = None
   467→    if property.location.parent:
   468→        county = property.location.parent
   469→        if county.parent:
   470→            county = county.parent
   471→    else:
